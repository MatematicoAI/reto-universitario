match /teacherInvites/{teacherEmail} {
  function isAdmin() {
    return request.auth != null &&
      request.auth.uid == 'cypIxKkYtFhzYn7MYp5boB0OEJ62';
  }
  function isInstitutional() {
    return teacherEmail.matches('^[^@]+@cientifica[.]edu[.]pe$');
  }
  allow get: if isAdmin() || (
    request.auth != null &&
    request.auth.token.email_verified == true &&
    request.auth.token.email == teacherEmail &&
    isInstitutional()
  );
  allow list: if false;
  allow create, update: if isAdmin() && isInstitutional() &&
    request.resource.data.email == teacherEmail &&
    request.resource.data.active is bool;
  allow delete: if isAdmin();
}
